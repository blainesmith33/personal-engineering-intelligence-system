    BLOCKED_HUMAN = "BLOCKED_HUMAN"

        StopState.HALT_CONFLICT,
        StopState.HALT_STRUCTURAL,

from __future__ import annotations

import hashlib
from typing import Mapping, Any


class IdError(ValueError):
    """Raised when ID generation invariants are violated."""


# -------------------------------------------------------
# Core ID Generation
# -------------------------------------------------------


def _stable_serialize(data: Mapping[str, Any]) -> bytes:
    """
    Deterministic serialization (order-stable).
    """
    if not isinstance(data, Mapping):
        raise IdError("data must be a mapping")

    items = sorted(data.items(), key=lambda x: x[0])

    parts: list[str] = []

    for key, value in items:
        if isinstance(value, Mapping):
            value_str = _stable_serialize(value).decode()
        else:
            value_str = repr(value)

        parts.append(f"{key}={value_str}")

    return "|".join(parts).encode("utf-8")


def _blake3_like_hash(data: bytes) -> str:
    """
    Using SHA256 as a stand-in for deterministic hashing.
    """
    return hashlib.sha256(data).hexdigest()


def make_artifact_id(
    *,
    prefix: str,
    payload: Mapping[str, Any],
) -> str:
    """
    Deterministic artifact ID.

    prefix examples:
    - decision
    - evaluation
    - plan

    ID format:
    <prefix>_<hash>
    """
    if not isinstance(prefix, str) or not prefix:
        raise IdError("prefix must be a non-empty string")

    serialized = _stable_serialize(payload)
    digest = _blake3_like_hash(serialized)

    return f"{prefix}_{digest[:16]}"


# -------------------------------------------------------
# Specific ID Builders
# -------------------------------------------------------


def make_decision_id(payload: Mapping[str, Any]) -> str:
    return make_artifact_id(prefix="decision", payload=payload)


def make_evaluation_id(payload: Mapping[str, Any]) -> str:
    return make_artifact_id(prefix="evaluation", payload=payload)


def make_plan_id(payload: Mapping[str, Any]) -> str:
    return make_artifact_id(prefix="plan", payload=payload)

from __future__ import annotations

import json
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Mapping

from peis.core.validation import require_mapping, require_non_empty_str


class StorageError(ValueError):
    """Raised when storage invariants are violated."""


@dataclass(frozen=True, slots=True)
class StoredArtifactRef:
    """
    Immutable reference to a persisted artifact.

    This is the canonical return surface for storage writes.
    """

    artifact_id: str
    artifact_type: str
    path: str

    def __post_init__(self) -> None:
        require_non_empty_str(self.artifact_id, field_name="artifact_id")
        require_non_empty_str(self.artifact_type, field_name="artifact_type")
        require_non_empty_str(self.path, field_name="path")

    def to_dict(self) -> dict[str, Any]:
        return {
            "artifact_id": self.artifact_id,
            "artifact_type": self.artifact_type,
            "path": self.path,
        }

    @classmethod
    def from_dict(cls, data: Mapping[str, Any]) -> "StoredArtifactRef":
        require_mapping(data, field_name="data")
        return cls(
            artifact_id=require_non_empty_str(
                data.get("artifact_id"),
                field_name="artifact_id",
            ),
            artifact_type=require_non_empty_str(
                data.get("artifact_type"),
                field_name="artifact_type",
            ),
            path=require_non_empty_str(
                data.get("path"),
                field_name="path",
            ),
        )


def store_artifact(
    *,
    root_dir: str | Path,
    artifact_type: str,
    artifact_id: str,
    payload: Mapping[str, Any],
    overwrite: bool = False,
) -> StoredArtifactRef:
    """
    Persist a single artifact as JSON.

    Layout:
    <root_dir>/<artifact_type>/<artifact_id>.json

    Invariants:
    - artifact_type must be non-empty
    - artifact_id must be non-empty
    - payload must be a mapping
    - existing artifacts are immutable unless overwrite=True
    """
    root = _coerce_root_dir(root_dir)

    - plan_record

    if evaluation_record is None:

    if plan_record is None:
        raise StorageError("result.plan_record is required")

    value = getattr(obj, attr_name, None)
    return require_non_empty_str(value, field_name=field_name)

    to_dict = getattr(obj, "to_dict", None)
    if not callable(to_dict):
        raise StorageError("record must expose a callable to_dict()")

    payload = to_dict()
    require_mapping(payload, field_name="payload")
    return dict(payload)

    with path.open("w", encoding="utf-8") as handle:
        json.dump(
            payload,
            handle,
            indent=2,
            sort_keys=True,
            ensure_ascii=False,
        )
        handle.write("\n")

import json

class StorageLoadError(ValueError):
    """Raised when stored artifacts cannot be loaded safely."""


def load_artifact(
    *,
    root_dir: str | Path,
    artifact_type: str,
    artifact_id: str,
) -> dict[str, Any]:
    """
    Load a single stored artifact JSON payload.

    Expected layout:
    <root_dir>/<artifact_type>/<artifact_id>.json
    """
    root = _coerce_root_dir(root_dir)
    artifact_type = require_non_empty_str(
        artifact_type,
        field_name="artifact_type",
    )
    artifact_id = require_non_empty_str(
        artifact_id,
        field_name="artifact_id",
    )

    target_path = root / artifact_type / f"{artifact_id}.json"

    if not target_path.exists():
        raise StorageLoadError(f"artifact not found: {target_path}")

    if not target_path.is_file():
        raise StorageLoadError(f"artifact path is not a file: {target_path}")

    try:
        with target_path.open("r", encoding="utf-8") as handle:
            payload = json.load(handle)
    except json.JSONDecodeError as exc:
        raise StorageLoadError(
            f"artifact JSON is invalid: {target_path}"
        ) from exc

    require_mapping(payload, field_name="payload")
    return dict(payload)


def load_decision_record(
    *,
    root_dir: str | Path,
    decision_record_id: str,
) -> DecisionRecord:
    payload = load_artifact(
        root_dir=root_dir,
        artifact_type="decision_record",
        artifact_id=decision_record_id,
    )

    try:
        return DecisionRecord.from_dict(payload)
    except Exception as exc:
        raise StorageLoadError(
            f"failed to load DecisionRecord: {decision_record_id}"

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Callable, Dict, Mapping

from peis.core.validation import require_mapping, require_non_empty_str


class ExecutionRegistryError(ValueError):
    """Raised when registry invariants are violated."""


# -------------------------------------------------------
# Capability Definition
# -------------------------------------------------------


@dataclass(frozen=True, slots=True)
class Capability:
    """
    A single executable capability.

    This is the ONLY allowed execution surface.
    """

    action_type: str
    handler: Callable[[Mapping[str, Any]], Mapping[str, Any]]
    requires_human_authorization: bool = False
    description: str | None = None

    def __post_init__(self) -> None:
        require_non_empty_str(self.action_type, field_name="action_type")

        if not callable(self.handler):
            raise ExecutionRegistryError("handler must be callable")

        if self.description is not None:
            require_non_empty_str(
                self.description,
                field_name="description",
            )


# -------------------------------------------------------
# Registry
# -------------------------------------------------------


class CapabilityRegistry:
    """
    Deterministic registry of allowed execution capabilities.

    Properties:
    - Explicit allow-list only
    - No dynamic imports
    - No fallback execution
    - No implicit behavior
    """

    def __init__(self) -> None:
        self._capabilities: Dict[str, Capability] = {}

    def register(self, capability: Capability) -> None:
        if not isinstance(capability, Capability):
            raise ExecutionRegistryError(
                "capability must be a Capability"
            )

        action_type = capability.action_type

        if action_type in self._capabilities:
            raise ExecutionRegistryError(
                f"duplicate capability registration: {action_type}"
            )

        self._capabilities[action_type] = capability

    def get(self, action_type: str) -> Capability:
        action_type = require_non_empty_str(
            action_type,
            field_name="action_type",
        )

        capability = self._capabilities.get(action_type)

        if capability is None:
            raise ExecutionRegistryError(
                f"unknown action_type: {action_type}"
            )

        return capability

    def has(self, action_type: str) -> bool:
        return action_type in self._capabilities

    def list_action_types(self) -> tuple[str, ...]:
        return tuple(sorted(self._capabilities.keys()))

    def to_dict(self) -> dict[str, Any]:
        return {
            "capabilities": [
                {
                    "action_type": c.action_type,
                    "requires_human_authorization": c.requires_human_authorization,
                    "description": c.description,
                }
                for c in self._capabilities.values()
            ]
        }


# -------------------------------------------------------
# Execution
# -------------------------------------------------------


def execute_capability(
    *,
    registry: CapabilityRegistry,
    action_type: str,
    payload: Mapping[str, Any],
    human_authorized: bool,
) -> dict[str, Any]:
    """
    Execute a registered capability.

    Enforces:
    - capability existence
    - payload validity
    - human authorization (if required)
    """
    if not isinstance(registry, CapabilityRegistry):
        raise ExecutionRegistryError("registry must be a CapabilityRegistry")

    capability = registry.get(action_type)

    require_mapping(payload, field_name="payload")

    if capability.requires_human_authorization and not human_authorized:
        raise ExecutionRegistryError(
            f"capability requires human authorization: {action_type}"
        )

    try:
        result = capability.handler(payload)
    except Exception as exc:
        raise ExecutionRegistryError(
            f"capability execution failed: {action_type}"
        ) from exc

    require_mapping(result, field_name="result")

    return dict(result)


# -------------------------------------------------------
# Default Safe Capability (for testing / baseline)
# -------------------------------------------------------


def noop_handler(payload: Mapping[str, Any]) -> Mapping[str, Any]:
    """
    Safe baseline capability.

    Does nothing except echo payload.
    """
    require_mapping(payload, field_name="payload")
    return {"status": "noop", "input": dict(payload)}


def build_default_registry() -> CapabilityRegistry:
    """
    Minimal registry with safe baseline capabilities.
    """
    registry = CapabilityRegistry()

    registry.register(
        Capability(
            action_type="noop",
            handler=noop_handler,
            requires_human_authorization=False,
            description="No-op capability for testing",
        )
    )

    return registry

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Mapping

from peis.core.validation import require_bool, require_mapping, require_non_empty_str
from peis.execution.registry import (
    CapabilityRegistry,
    ExecutionRegistryError,
    execute_capability,
)
from peis.plan_record import PlanRecord, PlanRecordError, PlanStep


class ExecutionError(ValueError):
    """Raised when execution invariants are violated."""


@dataclass(frozen=True, slots=True)
class StepExecutionResult:
    """
    Immutable result for one executed plan step.
    """

    step_id: str
    action_type: str
    status: str
    output: Mapping[str, Any]

    def __post_init__(self) -> None:
        require_non_empty_str(self.step_id, field_name="step_id")
        require_non_empty_str(self.action_type, field_name="action_type")
        require_non_empty_str(self.status, field_name="status")
        require_mapping(self.output, field_name="output")

    def to_dict(self) -> dict[str, Any]:
        return {
            "step_id": self.step_id,
            "action_type": self.action_type,
            "status": self.status,
            "output": dict(self.output),
        }


@dataclass(frozen=True, slots=True)
class PlanExecutionResult:
    """
    Immutable result for an executed plan.
    """

    plan_record_id: str
    subject_id: str
    executed: bool
    step_results: tuple[StepExecutionResult, ...]

    def __post_init__(self) -> None:
        require_non_empty_str(self.plan_record_id, field_name="plan_record_id")
        require_non_empty_str(self.subject_id, field_name="subject_id")
        require_bool(self.executed, field_name="executed")

        if not isinstance(self.step_results, tuple):
            raise ExecutionError(
                "step_results must be a tuple[StepExecutionResult, ...]"
            )

        for index, result in enumerate(self.step_results):
            if not isinstance(result, StepExecutionResult):
                raise ExecutionError(
                    f"step_results[{index}] must be a StepExecutionResult"
                )

    @property
    def is_noop_execution(self) -> bool:
        return len(self.step_results) == 0

    def to_dict(self) -> dict[str, Any]:
        return {
            "plan_record_id": self.plan_record_id,
            "subject_id": self.subject_id,
            "executed": self.executed,
            "step_results": [result.to_dict() for result in self.step_results],
        }


def execute_plan_record(
    *,
    registry: CapabilityRegistry,
    plan_record: PlanRecord,
) -> PlanExecutionResult:
    """
    Execute a PlanRecord deterministically through the registered capability surface.

    Enforces:
    - plan_record must be valid
    - only registered capabilities may execute
    - per-step human authorization is enforced by both the PlanStep and capability
    - empty-step plans are valid no-op executions
    """
    if not isinstance(registry, CapabilityRegistry):
        raise ExecutionError("registry must be a CapabilityRegistry")

    if not isinstance(plan_record, PlanRecord):
        raise ExecutionError("plan_record must be a PlanRecord")

    step_results: list[StepExecutionResult] = []

    for index, step in enumerate(plan_record.steps):
        if not isinstance(step, PlanStep):
            raise ExecutionError(f"plan_record.steps[{index}] must be a PlanStep")

        payload = _build_step_payload(plan_record=plan_record, step=step)

        try:
            output = execute_capability(
                registry=registry,
                action_type=step.action_type,
                payload=payload,
                human_authorized=plan_record.human_authorized,
            )
        except ExecutionRegistryError as exc:
            raise ExecutionError(
                f"step execution failed for step_id={step.step_id}: {exc}"
            ) from exc

        step_results.append(
            StepExecutionResult(
                step_id=step.step_id,
                action_type=step.action_type,
                status=_extract_status(output),
                output=output,
            )
        )

    return PlanExecutionResult(
        plan_record_id=plan_record.plan_record_id,
        subject_id=plan_record.subject_id,
        executed=True,
        step_results=tuple(step_results),
    )


def _build_step_payload(
    *,
    plan_record: PlanRecord,
    step: PlanStep,
) -> dict[str, Any]:
    metadata: Mapping[str, Any]
    if step.metadata is None:
        metadata = {}
    else:
        require_mapping(step.metadata, field_name="step.metadata")
        metadata = step.metadata

    return {
        "plan_record_id": plan_record.plan_record_id,
        "subject_id": plan_record.subject_id,
        "plan_type": plan_record.plan_type,
        "basis_decision_record_id": plan_record.basis_decision_record_id,
        "basis_stop_state": plan_record.basis_stop_state.value,
        "step_id": step.step_id,
        "action_type": step.action_type,
        "description": step.description,
        "requires_human_authorization": step.requires_human_authorization,
        "metadata": dict(metadata),
    }


def _extract_status(output: Mapping[str, Any]) -> str:
    value = output.get("status")
    if not isinstance(value, str) or not value.strip():
        return "completed"
    return value.strip()

from typing import Any, Callable, Dict, Mapping

from peis.core.validation import (

    require_mapping,
    require_non_empty_str,
)


class ExecutionRegistryError(ValueError):
    """Raised when registry invariants are violated."""


# -------------------------------------------------------
# Capability Definition
# -------------------------------------------------------


@dataclass(frozen=True, slots=True)
class Capability:
    """
    A single executable capability.

    This is the ONLY allowed execution surface.
    """

    action_type: str
    handler: Callable[[Mapping[str, Any]], Mapping[str, Any]]
    requires_human_authorization: bool = False
    description: str | None = None

    def __post_init__(self) -> None:
        require_non_empty_str(self.action_type, field_name="action_type")

        if not callable(self.handler):
            raise ExecutionRegistryError("handler must be callable")

        if self.description is not None:
            require_non_empty_str(
                self.description,
                field_name="description",
            )


# -------------------------------------------------------
# Registry
# -------------------------------------------------------


class CapabilityRegistry:
    """
    Deterministic registry of allowed execution capabilities.

    Properties:
    - Explicit allow-list only
    - No dynamic imports
    - No fallback execution
    - No implicit behavior
    """

    def __init__(self) -> None:
        self._capabilities: Dict[str, Capability] = {}

    def register(self, capability: Capability) -> None:
        if not isinstance(capability, Capability):
            raise ExecutionRegistryError(
                "capability must be a Capability"
            )

        action_type = capability.action_type

        if action_type in self._capabilities:
            raise ExecutionRegistryError(
                f"duplicate capability registration: {action_type}"
            )

        self._capabilities[action_type] = capability

    def get(self, action_type: str) -> Capability:
        action_type = require_non_empty_str(
            action_type,
            field_name="action_type",
        )

        capability = self._capabilities.get(action_type)

        if capability is None:
            raise ExecutionRegistryError(
                f"unknown action_type: {action_type}"
            )

        return capability

    def has(self, action_type: str) -> bool:
        return action_type in self._capabilities

    def list_action_types(self) -> tuple[str, ...]:
        return tuple(sorted(self._capabilities.keys()))

    def to_dict(self) -> dict[str, Any]:
        return {
            "capabilities": [
                {
                    "action_type": c.action_type,
                    "requires_human_authorization": c.requires_human_authorization,
                    "description": c.description,
                }
                for c in self._capabilities.values()
            ]
        }


# -------------------------------------------------------
# Execution
# -------------------------------------------------------


def execute_capability(
    *,
    registry: CapabilityRegistry,
    action_type: str,
    payload: Mapping[str, Any],
    human_authorized: bool,
) -> dict[str, Any]:
    """
    Execute a registered capability.

    Enforces:
    - capability existence
    - payload validity
    - human authorization (if required)
    - mapping-shaped result surface
    """
    if not isinstance(registry, CapabilityRegistry):
        raise ExecutionRegistryError("registry must be a CapabilityRegistry")

    capability = registry.get(action_type)

    try:
        require_mapping(payload, field_name="payload")
    except ValidationError as exc:
        raise ExecutionRegistryError("payload must be a mapping") from exc

    if capability.requires_human_authorization and not human_authorized:
        raise ExecutionRegistryError(
            f"capability requires human authorization: {action_type}"
        )

    try:
        result = capability.handler(payload)
    except Exception as exc:
        raise ExecutionRegistryError(
            f"capability execution failed: {action_type}"
        ) from exc

    try:
        require_mapping(result, field_name="result")
    except ValidationError as exc:
        raise ExecutionRegistryError(
            f"capability returned non-mapping result: {action_type}"
        ) from exc

    return dict(result)

# -------------------------------------------------------


def noop_handler(payload: Mapping[str, Any]) -> Mapping[str, Any]:
    """
    Safe baseline capability.

    Does nothing except echo payload.
    """
    require_mapping(payload, field_name="payload")
    return {"status": "noop", "input": dict(payload)}

            requires_human_authorization=False,

    )
