```python
from __future__ import annotations

import json
import hashlib
from dataclasses import dataclass, field
from typing import Any, Dict, List, Mapping, Optional


SCHEMA_VERSION = "1.0"
ARTIFACT_TYPE = "execution_result_record"


# =========================
# Deterministic Utilities
# =========================

def _canonicalize(obj: Any) -> Any:
    """
    Recursively canonicalize data into a deterministic structure.
    - Sort dict keys
    - Ensure lists are canonicalized element-wise
    """
    if isinstance(obj, Mapping):
        return {k: _canonicalize(obj[k]) for k in sorted(obj.keys())}
    elif isinstance(obj, list):
        return [_canonicalize(v) for v in obj]
    else:
        return obj


def _canonical_json(obj: Any) -> str:
    return json.dumps(_canonicalize(obj), separators=(",", ":"), ensure_ascii=False)


def make_execution_result_id(payload: Mapping[str, Any]) -> str:
    """
    Deterministic ID based on canonicalized payload (excluding ID field itself).
    """
    canonical = _canonical_json(payload)
    digest = hashlib.sha256(canonical.encode("utf-8")).hexdigest()
    return f"execution_result_{digest}"


# =========================
# Validation
# =========================

_ALLOWED_PLAN_STATUS = {"SUCCEEDED", "PARTIAL", "FAILED", "SKIPPED"}
_ALLOWED_STEP_STATUS = {"SUCCEEDED", "FAILED", "SKIPPED"}


def _require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def _validate_mapping(name: str, value: Any) -> None:
    _require(isinstance(value, Mapping), f"{name} must be a mapping")


def _validate_optional_mapping(name: str, value: Any) -> None:
    if value is not None:
        _validate_mapping(name, value)


def _validate_step_result(step: Mapping[str, Any], index: int) -> None:
    required_fields = {
        "step_index",
        "step_id",
        "action_type",
        "status",
        "input_payload",
        "output_payload",
        "error",
    }

    _validate_mapping("step_result", step)

    missing = required_fields - set(step.keys())
    _require(not missing, f"step_result[{index}] missing fields: {missing}")

    _require(step["step_index"] == index, "step_index must match list position")

    _require(
        step["status"] in _ALLOWED_STEP_STATUS,
        f"Invalid step status: {step['status']}"
    )

    _validate_mapping("input_payload", step["input_payload"])
    _validate_optional_mapping("output_payload", step["output_payload"])

    error = step["error"]
    if error is not None:
        _validate_mapping("error", error)
        _require("error_type" in error, "error missing error_type")
        _require("message" in error, "error missing message")


def _validate_execution_status(value: str) -> None:
    _require(value in _ALLOWED_PLAN_STATUS, f"Invalid execution_status: {value}")


# =========================
# Core Artifact
# =========================

@dataclass(frozen=True)
class ExecutionResultRecord:
    schema_version: str
    execution_result_record_id: str
    artifact_type: str

    authority: Mapping[str, Any]
    evidence: Mapping[str, Any]
    determinism: Mapping[str, Any]

    plan_record_id: str
    decision_record_id: str
    subject_id: str

    execution_status: str
    step_results: List[Mapping[str, Any]]
    summary: Mapping[str, Any]

    rationale: Optional[str] = None
    emitted_by: Optional[str] = None

    # -------------------------
    # Validation
    # -------------------------

    def __post_init__(self):
        _require(self.schema_version == SCHEMA_VERSION, "Invalid schema_version")
        _require(self.artifact_type == ARTIFACT_TYPE, "Invalid artifact_type")

        _validate_mapping("authority", self.authority)
        _validate_mapping("evidence", self.evidence)
        _validate_mapping("determinism", self.determinism)

        _require(self.plan_record_id, "plan_record_id required")
        _require(self.decision_record_id, "decision_record_id required")
        _require(self.subject_id, "subject_id required")

        _validate_execution_status(self.execution_status)

        _require(isinstance(self.step_results, list), "step_results must be a list")
        for idx, step in enumerate(self.step_results):
            _validate_step_result(step, idx)

        _validate_mapping("summary", self.summary)

    # -------------------------
    # Serialization
    # -------------------------

    def to_dict(self) -> Dict[str, Any]:
        return {
            "schema_version": self.schema_version,
            "execution_result_record_id": self.execution_result_record_id,
            "artifact_type": self.artifact_type,
            "authority": self.authority,
            "evidence": self.evidence,
            "determinism": self.determinism,
            "plan_record_id": self.plan_record_id,
            "decision_record_id": self.decision_record_id,
            "subject_id": self.subject_id,
            "execution_status": self.execution_status,
            "step_results": self.step_results,
            "summary": self.summary,
            "rationale": self.rationale,
            "emitted_by": self.emitted_by,
        }

    @classmethod
    def from_dict(cls, data: Mapping[str, Any]) -> "ExecutionResultRecord":
        return cls(**data)


# =========================
# Builder
# =========================

def build_execution_result_record(
    *,
    authority: Mapping[str, Any],
    evidence: Mapping[str, Any],
    determinism: Mapping[str, Any],
    plan_record_id: str,
    decision_record_id: str,
    subject_id: str,
    execution_status: str,
    step_results: List[Mapping[str, Any]],
    summary: Mapping[str, Any],
    rationale: Optional[str] = None,
    emitted_by: Optional[str] = None,
) -> ExecutionResultRecord:
    """
    Deterministic builder.
    ID is derived from canonical payload excluding ID field.
    """

    base_payload = {
        "schema_version": SCHEMA_VERSION,
        "artifact_type": ARTIFACT_TYPE,
        "authority": authority,
        "evidence": evidence,
        "determinism": determinism,
        "plan_record_id": plan_record_id,
        "decision_record_id": decision_record_id,
        "subject_id": subject_id,
        "execution_status": execution_status,
        "step_results": step_results,
        "summary": summary,
        "rationale": rationale,
        "emitted_by": emitted_by,
    }

    execution_result_record_id = make_execution_result_id(base_payload)

    return ExecutionResultRecord(
        schema_version=SCHEMA_VERSION,
        execution_result_record_id=execution_result_record_id,
        artifact_type=ARTIFACT_TYPE,
        authority=authority,
        evidence=evidence,
        determinism=determinism,
        plan_record_id=plan_record_id,
        decision_record_id=decision_record_id,
        subject_id=subject_id,
        execution_status=execution_status,
        step_results=step_results,
        summary=summary,
        rationale=rationale,
        emitted_by=emitted_by,
    )
```

```python
    - per-step human authorization is enforced by both the PlanStep and capability
    - execution stops on first failure
    - return value is always an ExecutionResultRecord

        raise ExecutionError("registry must be a CapabilityRegistry")

    for index, step in enumerate(plan_record.steps):
        if not isinstance(step, PlanStep):
            raise ExecutionError(f"plan_record.steps[{index}] must be a PlanStep")

        input_payload = _build_step_payload(plan_record=plan_record, step=step)

                    "action_type": step.action_type,

                }

                    "step_index": index,

                    "status": "FAILED",

                    "output_payload": None,
                    "error": {
                        "error_type": type(exc).__name__,
                        "message": str(exc),
                    },
                }
            )

def _build_step_payload(
    *,
    plan_record: PlanRecord,
    step: PlanStep,
) -> dict[str, Any]:
    metadata: Mapping[str, Any]
    if step.metadata is None:
        metadata = {}
    else:
        require_mapping(step.metadata, field_name="step.metadata")
        metadata = step.metadata

    return {
        "plan_record_id": plan_record.plan_record_id,
        "subject_id": plan_record.subject_id,
        "plan_type": plan_record.plan_type,
        "basis_decision_record_id": plan_record.basis_decision_record_id,
        "basis_stop_state": plan_record.basis_stop_state.value,
        "step_id": step.step_id,
        "action_type": step.action_type,
        "description": step.description,
        "requires_human_authorization": step.requires_human_authorization,
        "metadata": dict(metadata),
    }


def _derive_execution_status(step_results: list[Mapping[str, Any]]) -> str:
    if not step_results:
        return "SKIPPED"

    if all(step["status"] == "SUCCEEDED" for step in step_results):
        return "SUCCEEDED"

    if step_results[0]["status"] == "FAILED":
        return "FAILED"

    return "PARTIAL"

    return {
        "steps_total": len(step_results),
        "steps_succeeded": sum(1 for step in step_results if step["status"] == "SUCCEEDED"),
        "steps_failed": sum(1 for step in step_results if step["status"] == "FAILED"),
        "steps_skipped": sum(1 for step in step_results if step["status"] == "SKIPPED"),
    }


def _extract_required_mapping(obj: Any, field_name: str) -> Mapping[str, Any]:
    value = getattr(obj, field_name, None)
    require_mapping(value, field_name=field_name)
    return value


def _extract_required_non_empty_str(obj: Any, field_name: str) -> str:
    value = getattr(obj, field_name, None)
    if not isinstance(value, str) or not value.strip():
        raise ExecutionError(f"{field_name} must be a non-empty string")
    return value.strip()

    if isinstance(value, str) and value.strip():
        return value.strip()

    fallback = getattr(plan_record, "basis_decision_record_id", None)
    if isinstance(fallback, str) and fallback.strip():
        return fallback.strip()

    raise ExecutionError(
        "plan_record must provide decision_record_id or basis_decision_record_id"
    )
```

```python
) -> ExecutionResultRecord:
    """
    Load an ExecutionResultRecord from storage.
    """
    if not isinstance(execution_result_record_id, str) or not execution_result_record_id.strip():
        raise ValueError("execution_result_record_id must be a non-empty string")

    root_path = Path(root)
    file_path = root_path / "execution_result_record" / f"{execution_result_record_id}.json"

    if not file_path.exists():
        raise FileNotFoundError(f"ExecutionResultRecord not found: {file_path}")

    with file_path.open("r", encoding="utf-8") as f:
        payload = json.load(f)

    return ExecutionResultRecord.from_dict(payload)
```

```python
from __future__ import annotations

import argparse
import json
from pathlib import Path

from peis.execution.executor import execute_plan_record
from peis.execution.persist import persist_execution_result
from peis.execution.registry import CapabilityRegistry, Capability
from peis.execution.result_record import build_execution_result_record
from peis.plan_record import PlanRecord


def main() -> None:
    parser = argparse.ArgumentParser(description="Execute a PlanRecord and persist results.")
    parser.add_argument("--plan", required=True, help="Path to plan_record.json")
    parser.add_argument("--out", required=True, help="Output root directory")

    args = parser.parse_args()

    plan_path = Path(args.plan)
    out_root = Path(args.out)

    if not plan_path.exists():
        raise SystemExit(f"Plan file not found: {plan_path}")

    plan_record = _load_plan(plan_path)

    registry = CapabilityRegistry()
    _register_default_capabilities(registry)

    plan_result = execute_plan_record(
        registry=registry,
        plan_record=plan_record,
    )

    record = _convert_to_execution_result_record(plan_record, plan_result)

    file_path = persist_execution_result(
        root=out_root,
        record=record,
    )

    print(f"ExecutionResultRecord written to: {file_path}")


def _load_plan(path: Path) -> PlanRecord:
    with path.open("r", encoding="utf-8") as f:
        payload = json.load(f)

    return PlanRecord.from_dict(payload)


def _register_default_capabilities(registry: CapabilityRegistry) -> None:
    import subprocess

    def noop_handler(payload):
        return {"status": "ok"}

    def run_pytest_handler(payload):
        result = subprocess.run(
            ["python", "-m", "pytest", "-q"],
            capture_output=True,
            text=True,
        )
        return {
            "status": "ok" if result.returncode == 0 else "failed",
            "stdout": result.stdout,
            "stderr": result.stderr,
        }

    registry.register(
        Capability(
            action_type="noop",
            handler=noop_handler,
        )
    )

    registry.register(
        Capability(
            action_type="run_pytest",
            handler=run_pytest_handler,
        )
    )


def _convert_to_execution_result_record(plan_record, plan_result):
    step_results = []

    for idx, step in enumerate(plan_result.step_results):
        step_results.append(
            {
                "step_index": idx,
                "step_id": step.step_id,
                "action_type": step.action_type,
                "status": "SUCCEEDED",
                "input_payload": {},
                "output_payload": step.output,
                "error": None,
            }
        )

    return build_execution_result_record(
        authority=plan_record.authority.to_dict(),
        evidence=plan_record.evidence.to_dict(),
        determinism=plan_record.determinism.to_dict(),
        plan_record_id=plan_record.plan_record_id,
        decision_record_id=plan_record.basis_decision_record_id,
        subject_id=plan_record.subject_id,
        execution_status="SUCCEEDED",
        step_results=step_results,
        summary={
            "steps_total": len(step_results),
            "steps_succeeded": len(step_results),
            "steps_failed": 0,
            "steps_skipped": 0,
        },
        emitted_by="cli",
    )


if __name__ == "__main__":
    main()
```
